cmake_minimum_required(VERSION 3.21)
project(pdf_reader VERSION 0.1.1 LANGUAGES CXX)

include(FetchContent)
option(PDF_READER_USE_LOCAL_MWFL "Use a local mwfl checkout" OFF)
set(PDF_READER_MWFL_SOURCE_DIR "../mwfl" CACHE PATH "Local mwfl source directory")
set(MWFL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MWFL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MWFL_BUILD_WEBVIEW2 ON CACHE BOOL "" FORCE)
if(PDF_READER_USE_LOCAL_MWFL)
    add_subdirectory("${PDF_READER_MWFL_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/mwfl")
else()
  FetchContent_Declare(mwfl GIT_REPOSITORY https://github.com/mwfl/mwfl.git GIT_TAG v0.3.0)
    FetchContent_MakeAvailable(mwfl)
endif()
add_executable(pdf-reader WIN32
    main.cpp pdf_reader.rc resource.h app-icon.ico app.manifest)
target_link_libraries(pdf-reader PRIVATE mwfl::app mwfl::webview2 shlwapi)
target_compile_features(pdf-reader PRIVATE cxx_std_20)
target_compile_definitions(pdf-reader PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX MWFL_APP_VERSION=L\"${PROJECT_VERSION}\")
if(MSVC)
    target_compile_options(pdf-reader PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endif()

enable_testing()
add_test(NAME pdf-reader.gui COMMAND pdf-reader --self-test)
set_tests_properties(pdf-reader.gui PROPERTIES TIMEOUT 45)

install(TARGETS pdf-reader RUNTIME DESTINATION . COMPONENT Application)
install(FILES README.md LICENSE DESTINATION . COMPONENT Application)
